from pathlib import Path
import hashlib,json,subprocess,sys,time
ROOT=Path('/private/tmp/vellum-grid-cell-editing')
FIX=Path('/private/tmp/vellum-grid-editing-qualifier-missing-file')
OUT=ROOT/'docs/reports/macos-grid-editing/qualifier-deletion-check'
assert not FIX.exists() and not OUT.exists()
sys.dont_write_bytecode=True
sys.path.insert(0,str(ROOT))
from scripts.child_process import run_child
sha=lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
subprocess.run(['git','clone','--no-hardlinks','--quiet',str(ROOT),str(FIX)],check=True)
build=FIX/'drivers/cogent/build'
if not build.exists():build.symlink_to((ROOT/'drivers/cogent/build').resolve(),target_is_directory=True)
relative='docs/reports/macos-grid-editing/qualify-core.py'
script=FIX/relative;script.parent.mkdir(parents=True,exist_ok=True)
original=subprocess.check_output(['git','show',':'+relative],cwd=ROOT)
corrected=(ROOT/relative).read_bytes()
(FIX/'ui/node.go').unlink()
OUT.mkdir(parents=True)
(OUT/'original-qualifier.py.txt').write_bytes(original)
(OUT/'corrected-qualifier.py.txt').write_bytes(corrected)
checks=[]
for name,payload in [('before',original),('after',corrected)]:
    script.write_bytes(payload)
    output=OUT/name
    log=OUT/(name+'-process.log')
    started=time.monotonic()
    with log.open('xb') as stream:
        argv=['python3',str(script),str(output)]
        result=run_child(argv,cwd=FIX,stdout=stream,stderr=subprocess.STDOUT,timeout=300)
    checks.append({'argv':argv,'exit_code':result.returncode,'elapsed_seconds':round(time.monotonic()-started,3),'log':log.name,'log_sha256':sha(log),'qualifier_sha256':sha(script),'qualification_exists':(output/'qualification.json').is_file()})
    print(json.dumps(checks[-1]),flush=True)
old,new=checks
q=json.loads((OUT/'after/qualification.json').read_text())
passed=old['exit_code']!=0 and not old['qualification_exists'] and new['exit_code']!=0 and new['qualification_exists'] and q['result']=='FAIL' and q['unchanged'] and q['source_inputs']['ui/node.go']=={'absent':True} and len(q['checks'])==1 and q['checks'][0]['exit_code']!=0
record={'result':'PASS' if passed else 'FAIL','scope':'Real isolated tracked ui/node.go deletion reproduces the original capture crash; corrected wrapper reaches Go compile refusal and writes an exclusive FAIL proof with the absent source marker. No candidate source changes, no fake Go runner, no GUI/performance claim.','base':subprocess.check_output(['git','rev-parse','HEAD'],cwd=FIX,text=True).strip(),'fixture':str(FIX),'removed_tracked_file':'ui/node.go','original_wrapper_sha256':hashlib.sha256(original).hexdigest(),'corrected_wrapper_sha256':hashlib.sha256(corrected).hexdigest(),'checks':checks,'failure_proof_sha256':sha(OUT/'after/qualification.json')}
(OUT/'regression.json').write_text(json.dumps(record,indent=2)+'\n')
print(json.dumps({'result':record['result'],'proof':str(OUT/'regression.json'),'sha256':sha(OUT/'regression.json')}),flush=True)
sys.exit(0 if passed else 1)
