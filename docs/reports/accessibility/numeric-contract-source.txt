package main
import (
 "encoding/json"
 "fmt"
 "os"
 "example.com/vellum/ui"
)
func main() {
 results:=map[string]string{}
 for _,kind:=range []ui.Kind{ui.Slider,ui.NumberField} {
  for label,numeric:=range map[string]ui.Numeric{"omitted-zero-bounds":{},"equal-zero-bounds":{Min:0,Max:0,Step:1},"zero-step":{Min:0,Max:10,Step:0}} {
   n:=ui.Node{ID:"numeric",Kind:kind,Numeric:numeric}
   if err:=ui.Validate(n);err==nil {fmt.Fprintln(os.Stderr,"unexpectedly admitted",kind,label);os.Exit(1)}
   results[string(kind)+":"+label]="refused"
  }
  n:=ui.Node{ID:"numeric",Kind:kind,Number:4,Numeric:ui.Numeric{Min:0,Max:10,Step:1}}
  if err:=ui.Validate(n);err!=nil {fmt.Fprintln(os.Stderr,err);os.Exit(1)}
  results[string(kind)+":declared-range"]="accepted"
 }
 json.NewEncoder(os.Stdout).Encode(map[string]any{"result":"PASS","mandatory_numeric_contract":results})
}
