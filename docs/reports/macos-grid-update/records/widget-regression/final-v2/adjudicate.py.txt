"""Reclassify standard t.Logf evidence lines without rerunning child tests."""
from pathlib import Path
import hashlib,json,re,sys
OUT=Path(__file__).resolve().parent
ROOT=Path('/private/tmp/vellum-large-data')
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
initial_path=OUT/'producer-initial-proof.json'
initial=json.loads(initial_path.read_text())
assert initial['unchanged'] and initial['assets_unchanged'] and initial['inherited_v1']['unchanged']
assert initial['red_counter_assertions']['actual_count']==1626
raster_lines=['GRID_UPDATE_RGBA '+record for record in initial['software_rgba']['original_records']]
raster_grammar=r'GRID_UPDATE_RGBA dpi=(96|120|144|192) phase=(replace-scroll-select|clear-and-move-cursor) bounds=\(-?\d+,-?\d+\)-\(-?\d+,-?\d+\) stride=\d+ sha256=[0-9a-f]{64}'
assert len(raster_lines)==8 and all(re.fullmatch(raster_grammar,line) for line in raster_lines)
assert initial['red_counter_assertions']['other_failures']==raster_lines
assert initial['software_rgba']['original_overlay_matches_v1'] and initial['software_rgba']['exact_original_match_each_repetition']
assert [c['exit_code'] for c in initial['checks']]==[1,0]
checks=initial['checks']
for c in checks:
    assert sha(Path(c['log']))==c['log_sha256']
red=Path(checks[0]['log']).read_text();green=Path(checks[1]['log']).read_text()
lines=re.findall(r'grid_update_offscreen_test\.go:\d+: ([^\n]*)',red)
assert [line for line in lines if line in raster_lines]==raster_lines
failures=[line for line in lines if line not in raster_lines]
assert len(failures)==1626 and all('retained widget update/style passes: got 2/2 want 1/1' in line for line in failures)
assert 'WARNING: DATA RACE' not in red+green
current={path:sha(ROOT/path) for path in initial['after']}
assert current==initial['before']==initial['after']
history={p.name:sha(p) for p in sorted(OUT.parent.iterdir()) if p.is_file()}
assert history==initial['inherited_v1']['history_before']==initial['inherited_v1']['history_after']
assets={name:sha(OUT/name) for name in initial['assets_after']}
assert assets==initial['assets_before']==initial['assets_after']
proof=dict(initial)
proof['red_counter_assertions']={'expected_count':1626,'actual_count':len(failures),'other_failures':[]}
proof['result']='PASS'
proof['producer_adjudication']={
 'initial_proof':{'path':str(initial_path),'sha256':sha(initial_path),'result':'FAIL'},
 'reason':'Initial producer parser counted all test-prefixed lines as assertion failures; eight standard t.Logf GRID_UPDATE_RGBA lines are successful evidence. Whitelisting only the exact8 known original raster records, each checked against its complete grammar, yields1626 expected2/2 assertions and zero other assertions. No source, rawlog, test result or child-process rerun was changed.',
 'script':{'path':str(Path(__file__)),'sha256':sha(Path(__file__))},
 'current_selected_inputs':current,'current_v1_history':history,'current_initial_assets':assets,
 'checks_rerun':False,
}
(OUT/'qualification.json').write_text(json.dumps(proof,indent=2)+'\n')
print(json.dumps({'result':'PASS','qualification':str(OUT/'qualification.json'),'sha256':sha(OUT/'qualification.json'),'test_sha256':current['drivers/cogent/grid_update_offscreen_test.go'],'grid_sha256':current['drivers/cogent/grid.go'],'red_assertions':len(failures),'other_failures':0,'pixel_matches':True},indent=2))
