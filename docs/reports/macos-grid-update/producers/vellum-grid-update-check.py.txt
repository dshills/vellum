#!/usr/bin/env python3
import hashlib
import json
import re
import subprocess
import time
from pathlib import Path

ROOT = Path('/private/tmp/vellum-large-data')
OUT = Path('/private/tmp/vellum-grid-update-qualification/affected-checks')
OUT.mkdir(exist_ok=False)

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

selected = ['drivers/cogent/grid.go', 'drivers/cogent/grid_update_offscreen_test.go', 'specs/grid-native.md']
source_before = {p: sha(ROOT / p) for p in selected}
tests = []
for path in sorted((ROOT / 'drivers/cogent').glob('*_test.go')):
    if path.name.startswith(('grid_', 'frame_metrics_')) and path.name != 'grid_update_offscreen_test.go':
        tests.extend(re.findall(r'^func (Test\w+)\(t \*testing.T\)', path.read_text(), re.M))
regexp = '^(' + '|'.join(tests) + ')$'
argv = ['go', 'test', '-race', '-tags', 'offscreen', '-count=1', '-run', regexp, '.']
start = time.monotonic()
with (OUT / 'native-grid-and-frame-race.log').open('xb') as raw:
    result = subprocess.run(argv, cwd=ROOT / 'drivers/cogent', stdout=raw, stderr=subprocess.STDOUT)
source_after = {p: sha(ROOT / p) for p in selected}
record = {'schema': 1, 'source_before': source_before, 'source_after': source_after,
          'scope': 'Affected existing managed grid, editing, selection, routing, namespace, rollback and frame-metrics offscreen race cases. New exact-count/pixel test separately qualified by its author. No desktop physical input or performance assertion.',
          'tests_selected': tests, 'argv': argv, 'exit_code': result.returncode,
          'elapsed_seconds': time.monotonic() - start,
          'log_sha256': sha(OUT / 'native-grid-and-frame-race.log'),
          'producer_sha256': sha(Path(__file__)),
          'result': 'PASS' if result.returncode == 0 and source_before == source_after else 'FAILED'}
(OUT / 'qualification.json').write_text(json.dumps(record, indent=2) + '\n')
print(record['result'], len(tests), result.returncode, flush=True)
raise SystemExit(0 if record['result'] == 'PASS' else 1)
