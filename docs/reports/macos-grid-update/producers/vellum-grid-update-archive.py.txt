#!/usr/bin/env python3
import hashlib
import json
from pathlib import Path

DEST = Path('/private/tmp/vellum-large-data/docs/reports/macos-grid-update')

def sha(data):
    return hashlib.sha256(data).hexdigest()

records = []
def copy(src, rel):
    data = src.read_bytes()
    data.decode('utf-8')  # Refuse binaries; preserve UTF-8 bytes exactly.
    target = DEST / rel
    target.parent.mkdir(parents=True, exist_ok=True)
    assert not target.exists(), target
    target.write_bytes(data)
    records.append({'original': str(src), 'archive': str(rel), 'bytes': len(data), 'sha256': sha(data)})

for original, relative in (
    (Path('/private/tmp/vellum-grid-update-qualification'), Path('records')),
    (Path('/private/tmp/vellum-grid-update-review'), Path('review/internal')),
    (Path('/private/tmp/vellum-grid-update-prism'), Path('review/prism')),
):
    assert original.is_dir(), original
    for source in sorted(original.rglob('*')):
        if not source.is_file():
            continue
        assert source.suffix in ('.json', '.log', '.txt', '.py', '.sh'), source
        rel = relative / source.relative_to(original)
        if source.suffix in ('.py', '.sh'):
            rel = rel.with_name(rel.name + '.txt')
        copy(source, rel)

for name in ('measure', 'check', 'prism', 'archive', 'audit', 'promote'):
    source = Path(f'/private/tmp/vellum-grid-update-{name}.py')
    copy(source, Path('producers') / (source.name + '.txt'))

manifest = {'schema': 1,
            'scope': 'Byte-preserved scoped qualification/build/regression/native raw records, independent review and eight-path Prism records. Absolute local paths are provenance. Inert producers/sources are text; binaries are excluded and identified by hash. No full compiler/module-cache/SDK closure or P6 pass.',
            'files': records}
(DEST / 'archive.json').write_text(json.dumps(manifest, indent=2) + '\n')
print('archived', len(records), sum(r['bytes'] for r in records), 'bytes', flush=True)
