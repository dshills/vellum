+# macOS grid input and workload qualification
+
+Stored-table warmup now refuses nonzero starting offsets, incomplete closed
+trajectories, extra callbacks and changed completed geometry/scale before ordinary
+measurements. A separate bounded scalar ledger makes admission, delivered wheel
+arithmetic and owner transitions inspectable. This advances reliable Mac
+performance qualification; it does not repair an established framework input
+ordering defect or claim a rendering speedup.
+
+## Implementation
+
+The opt-in `NewFrameMetricsWithGridInputDetails` constructor preallocates
+`3*tickets+256` records for at most4096 frame tickets. Ordinary metrics allocate
+no ledger or additional input clocks. Bindings copy identifiers up to4096 bytes.
+Records contain no model, widget, event, callback, cell values or native pointer.
+Exhaustion/oversized identifiers explicitly refuse diagnostic evidence; normal
+grid input continues with its existing behavior.
+
+Admissions copy logical requests, encoded physical deltas, completed scale and
+owner state. Actual native wheel handlers copy their current tag, handler scale,
+integer steps and signed remainder. Posted owner transitions reuse existing state
+reads and link the originating wheel sequence; scrollbar and layout paths remain
+distinct. Native style/remainder reads never move into a peer-window owner callback.
+An independent ledger mutex releases before posting and callouts. Admission
+appends happen after the metrics lock releases; tags/ingress sequences establish
+correlation, and append order is not FIFO or clock order. Tag zero means untraced,
+not desktop/user origin. A ledger snapshot is not a native barrier.
+
+`loadedgrid -perf-input-details=/new/path.json` saves the detached ledger after
+native Run retires, including failed warmups. Files are exclusive and capped at
+64MiB; overflow produces nonzero finalization. This diagnostic mode cannot combine
+with CPU/source profiling. Stage/sample flags agree and ordinary validation
+rejects diagnostic samples. Ordinary ready records now retain warm start/end
+viewports and generations plus callback count under `warmup_guard: 1`.
+See [the contract](../../../specs/grid-performance.md).

+// GridInputRecord contains copied scalar provenance, never an event, model,
+// widget or callback. Tag zero means untraced; it does not identify a producer.

+	RemainderBefore, RemainderAfter    [2]float64

+	GenerationBefore, GenerationAfter  uint64
+	ViewportBefore, ViewportAfter      virtual.GridViewport

+// NewFrameMetricsWithGridInputDetails records admissions, actual wheel
+// arithmetic and owner transitions, including untraced input and layout.
+// Extra synchronization makes these diagnostic samples ineligible for ordinary
+// latency qualification. The ordinary constructor allocates no input ledger.
+func NewFrameMetricsWithGridInputDetails(capacity int) (*FrameMetrics, error) {
+	if capacity < 1 || capacity > MaxSourceFrameMetricsCapacity {
+		return nil, ErrFrameMetricsConfig
+	}
+	m, err := newFrameMetrics(capacity, false)
+	if err != nil {
+		return nil, err
+	}
+	n := 3*capacity + 256
+	m.gridInputs = &gridInputRecorder{capacity: n, records: make([]GridInputRecord, 0, n)}
+	return m, nil
+}

+// GridInputDetails returns a detached snapshot. A full ledger refuses further
+// records and reports loss; it never overwrites evidence or alters grid input.
+func (m *FrameMetrics) GridInputDetails() GridInputDetails {
+	if !m.valid() {
+		return GridInputDetails{Failure: ErrFrameMetricsConfig.Error()}
+	}
+	r := m.gridInputs
+	if r == nil {
+		return GridInputDetails{}
+	}
+	r.mu.Lock()
+	defer r.mu.Unlock()
+	out := GridInputDetails{Capacity: r.capacity, Records: append([]GridInputRecord(nil), r.records...), Dropped: r.dropped, Failure: r.failure}
+	if r.dropped != 0 && out.Failure == "" {
+		out.Failure = "cogent: grid input details capacity exhausted"
+	}
+	return out
+}

+func bindGridInputDetails(g *gridBinding) *gridInputBinding {
+	frameMetricsRegistry.Lock()
+	b := frameMetricsRegistry.windows[frameWindowKey{g.owner, g.window}]
+	frameMetricsRegistry.Unlock()
+	if b == nil || b.metrics.gridInputs == nil {
+		return nil
+	}
+	if len(g.r.node.ID) > 4096 {
+		r := b.metrics.gridInputs
+		r.mu.Lock()
+		r.failure = "cogent: grid input identifier exceeds4096 bytes"
+		r.mu.Unlock()
+		return nil
+	}
+	return &gridInputBinding{recorder: b.metrics.gridInputs, window: strings.Clone(g.window), grid: strings.Clone(g.r.node.ID), incarnation: g.incarnation}
+}

+func (b *gridInputBinding) record(v GridInputRecord) uint64 {
+	if b == nil {
+		return 0
+	}
+	r := b.recorder
+	r.mu.Lock()
+	defer r.mu.Unlock()
+	if len(r.records) >= r.capacity {
+		if r.dropped != ^uint64(0) {
+			r.dropped++
+		}
+		return 0
+	}
+	v.Sequence = uint64(len(r.records)) + 1
+	v.Window, v.Grid, v.Incarnation = b.window, b.grid, b.incarnation
+	r.records = append(r.records, v)
+	return v.Sequence

+// O_EXCL preserves prior evidence and refuses existing files and symlinks.
+func startPerfInputDetails(path string, metrics *cogent.FrameMetrics) (func(error) error, error) {
+	if path == "" {
+		return func(error) error { return nil }, nil
+	}
+	if !validPerfProfilePath(path) {
+		return nil, errors.New("loadedgrid perf: invalid input details path")
+	}
+	f, err := os.OpenFile(path, os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0600)
+	if err != nil {
+		return nil, err
+	}
+	var once sync.Once
+	var terminal error
+	return func(runError error) error {
+		once.Do(func() {
+			result := "CAPTURED_DIAGNOSTIC"
+			if runError != nil {
+				result = "FAILED_RUN_DIAGNOSTIC"
+			}
+			details := metrics.GridInputDetails()
+			var detailError error
+			if details.Failure != "" {
+				result = "REFUSED_INPUT_DETAILS"
+				detailError = errors.New(details.Failure)
+			}
+			w := &perfProfileWriter{target: f, remaining: perfProfileMaxBytes}
+			err := json.NewEncoder(w).Encode(struct {
+				Schema, Result string
+				Details        cogent.GridInputDetails
+			}{"vellum-grid-input-v1", result, details})
+			terminal = errors.Join(detailError, err, w.failure(), f.Close())
+		})
+		return terminal
+	}, nil
