import hashlib
import json
import os
from pathlib import Path
import shutil
import subprocess
import time

root = Path('/private/tmp/vellum-grid-selection')
module = root / 'examples/grid'
out = Path('/private/tmp/vellum-grid-selection-example/qualified-v3')
out.mkdir(parents=True, exist_ok=False)
owned = ['source.go', 'model.go', 'main.go', 'smoke.go', 'README.md',
         'selection_rows_test.go', 'selection_rows_managed_test.go']
unchanged = ['model_test.go', 'managed_test.go', 'variable_rows_test.go', 'variable_rows_managed_test.go', 'go.mod', 'go.sum', 'LICENSE']
runtime = ['virtual/grid.go', 'virtual/grid_geometry.go', 'virtual/grid_rows.go', 'virtual/grid_selection.go',
           'drivers/cogent/grid.go', 'drivers/cogent/grid_geometry.go',
           'drivers/cogent/grid_accessibility.go', 'drivers/cogent/manual.go',
           'drivers/cogent/manual_offscreen.go', 'drivers/cogent/manual_desktop.go',
           'drivers/cogent/env.sh', 'drivers/cogent/go.mod',
           'drivers/cogent/native/dependencies.json']

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def inputs():
    return {
        'owned': {name: sha(module / name) for name in owned},
        'preserved_example': {name: sha(module / name) for name in unchanged},
        'integration_inputs': {name: sha(root / name) for name in runtime},
    }

before = inputs()
previous_path = Path('/private/tmp/vellum-grid-selection-example/qualified-v2/qualification.json')
previous = json.loads(previous_path.read_bytes())
assert before['owned'] == previous['after']['owned']
assert before['preserved_example'] == previous['after']['preserved_example']
native_path = Path('/private/tmp/vellum-grid-selection-native-owned-freeze-v2.json')
native = json.loads(native_path.read_bytes())
assert all(sha(root / name) == expected for name, expected in native['files'].items())
native_binding = {'path': str(native_path), 'sha256': sha(native_path), 'files': native['files']}
snapshot = out / 'owned-source'
snapshot.mkdir()
for name in owned:
    shutil.copyfile(module / name, snapshot / name)
(out / 'inputs-before.json').write_text(json.dumps(before, indent=2) + '\n')

commands = [
    ('selection-managed-race', "go test -tags offscreen -race -count=1 -run '^TestActualRowSelection' -v ./..."),
    ('managed-vet', 'go vet -tags offscreen ./...'),
    ('desktop-race-build', 'go build -race -o ' + str(out / 'grid-desktop') + ' .'),
]
prior_path = Path('/private/tmp/vellum-grid-selection-example/qualified-v1/qualification.json')
prior = json.loads(prior_path.read_bytes())
portable_owned = ['source.go', 'model.go', 'selection_rows_test.go']
portable_preserved = ['model_test.go', 'variable_rows_test.go', 'go.mod', 'go.sum']
portable_runtime = ['virtual/grid.go', 'virtual/grid_geometry.go', 'virtual/grid_rows.go', 'virtual/grid_selection.go']
assert all(before['owned'][name] == prior['after']['owned'][name] for name in portable_owned)
assert all(before['preserved_example'][name] == prior['after']['preserved_example'][name] for name in portable_preserved)
assert all(before['integration_inputs'][name] == prior['after']['integration_inputs'][name] for name in portable_runtime)
assert prior['checks'][0]['exit_code'] == 0 and sha(Path(prior['checks'][0]['log'])) == prior['checks'][0]['log_sha256']
results = [dict(prior['checks'][0], inherited_from=str(prior_path), inherited_proof_sha256=sha(prior_path), current_portable_inputs_unchanged=True)]
prefix = 'export VELLUM_CACHE_DIR=/tmp/vellum-p0-cache; . ' + str(root / 'drivers/cogent/env.sh') + '; '
for name, command in commands:
    log = out / (name + '.log')
    print('START', name, flush=True)
    started = time.monotonic()
    timed_out = False
    with log.open('wb') as stream:
        stream.write(('command: ' + prefix + command + '\n').encode())
        stream.flush()
        try:
            run = subprocess.run(['/bin/zsh', '-c', prefix + command], cwd=module,
                                 stdout=stream, stderr=subprocess.STDOUT, timeout=600)
            code = run.returncode
        except subprocess.TimeoutExpired:
            code, timed_out = 124, True
    result = {'name': name, 'command': prefix + command, 'cwd': str(module),
              'exit_code': code, 'timed_out': timed_out,
              'elapsed_seconds': round(time.monotonic() - started, 3),
              'log': str(log), 'log_sha256': sha(log)}
    results.append(result)
    print('END', json.dumps(result), flush=True)
    (out / 'checks.json').write_text(json.dumps(results, indent=2) + '\n')
    if code:
        break

after = inputs()
binary = out / 'grid-desktop'
proof = {
    'scope': 'Seven owned example files, seven preserved module/test files and thirteen selected integration inputs; v1 portable race20 inherited after exact relevant-input equality checks. Fresh affected row-selection actual-router race/vet/desktop build; old fixed/variable regression bodies unchanged and their v2 passed subtests remain historical. Native cursor geometry correction is bound separately. Not a complete build source closure or desktop execution claim.',
    'source_root': str(root), 'module': str(module),
    'native_freeze': native_binding,
    'previous_failed_proof': {'path': str(previous_path), 'sha256': sha(previous_path), 'consumer_inputs_unchanged': before['owned'] == previous['after']['owned']},
    'before': before, 'after': after, 'unchanged': before == after,
    'checks': results,
    'result': 'PASS' if before == after and all(sha(root / name) == expected for name, expected in native['files'].items()) and len(results) == len(commands) + 1 and all(r['exit_code'] == 0 for r in results) else 'FAIL',
    'binary': {'path': str(binary), 'sha256': sha(binary)} if binary.exists() else None,
    'exploratory_logs': {str(path): sha(path) for path in sorted(out.parent.glob('*initial.log')) + sorted(out.parent.glob('*followup.log'))},
    'desktop_invocation': str(binary) + ' -smoke -row-selection',
}
(out / 'qualification.json').write_text(json.dumps(proof, indent=2) + '\n')
print('PROOF', str(out / 'qualification.json'), sha(out / 'qualification.json'), proof['result'], flush=True)
raise SystemExit(0 if proof['result'] == 'PASS' else 1)
