+# Unchanged native grid refresh
+
+This checkpoint reuses a completed native grid projection when its validated
+model generation is unchanged. It also fixes a general retained-parent identity
+comparison that treated an embedded accessibility-host Frame as a different
+parent from the concrete native tree node on every refresh.
+
+The implementation keeps source-health checks, legitimate style/theme/topology
+updates, effective disabled and help-tooltip styling, constant-time canonical
+scrollbar restoration on refresh, and the private editor's independent
+selection/restore/preedit/focus lifecycle. An incomplete projection sync cannot
+be reused. Exact built-in worker projections validate their borrowed input stamp
+even when an empty or fixed-height view exposes unchanged constant metadata.
+Unstamped source mutations still violate the immutable source contract.
+
+## Correctness evidence
+
+The original grid/reconciliation overlay produces 1,497 expected regression
+assertions. The final candidate passes all five new native regression groups
+three times under the race detector with unchanged source and prepared inputs.
+These exercise actual managed refresh and completed-capture tails, zero Cell
+reads and retained update/style traversals, changed model projections,
+style/help/disabled/topology, private selection/restore/focus, and no-op/refused
+scrollbar restoration. The fixture uses the software backend; Manual still
+conservatively reports layoutPending after owner refresh, so this does not
+promise that public Capture immediately accepts a no-op refresh.
+
+A separate real embedded-host extension regression preserves the original
+Mount parent argument and Sync callbacks while eliminating fake parent moves,
+OnChildAdded and update/style work. Real style and parent changes still apply.
+Its original reconciliation overlay fails and the candidate passes. The full
+native adapter package passes the offscreen race suite and default/offscreen
+vet. All portable virtual tests pass under race and vet; the new projection
+health regression passes three times and its original overlay fails with 180
+expected assertions. Eight final software grid frames at 96/120/144/192 DPI
+match the previous checkpoint's 24 repeated pixel digests exactly.

+func watchRefreshGrid(g *gridBinding) *refreshGridCounts {
+	p := &refreshGridCounts{slots: map[*core.Text]*gridUpdatePass{}}
+	g.host.Updater(func() { p.host.updates++ })
+	g.host.FinalStyler(func(*styles.Style) { p.host.styles++ })
+	p.observeCurrentSlots(g)
+	return p

+// A changed projection may legitimately grow or shrink the pool. Establish its
+// accepted retained set before measuring only the following unchanged tail.
+func (p *refreshGridCounts) observeCurrentSlots(g *gridBinding) {
+	current := make(map[*core.Text]*gridUpdatePass, len(g.cells)+len(g.headers))
+	for _, w := range append(append([]*core.Text(nil), g.cells...), g.headers...) {
+		count := p.slots[w]
+		if count == nil {
+			count = new(gridUpdatePass)
+			w.Updater(func() { count.updates++ })
+			w.FinalStyler(func(*styles.Style) { count.styles++ })
+		}
+		current[w] = count
+	}
+	p.slots = current
+}
+func (p *refreshGridCounts) reset() {
+	p.host = gridUpdatePass{}
+	for _, slot := range p.slots {
+		*slot = gridUpdatePass{}
+	}
+}
+func (p *refreshGridCounts) assertZero(t *testing.T, label string, g *gridBinding) {

+func TestGridProjectionCurrentRejectsBorrowedInputDrift(t *testing.T) {
+	for variant, name := range []string{"fixed", "metrics", "edit", "metrics-edit"} {
+		for _, shape := range []string{"visible", "empty-input", "empty-filter"} {
+			for _, drift := range []string{"identity", "revision", "rows"} {
+				t.Run(fmt.Sprintf("%s/%s/%s", name, shape, drift), func(t *testing.T) {
+					rows := 3
+					if shape == "empty-input" {
+						rows = 0
+					}
+					base, source := projectionCurrentInput(t, rows, variant)
+					query := GridQuery{}
+					if shape == "empty-filter" {
+						query.Filters = []GridFilter{{ColumnID: "c:0", Kind: GridFilterEquals, Text: "not present"}}
+					}
+					g, view := projectionCurrentGrid(t, source, query)
+					before, err := g.State()
+					if err != nil {
+						t.Fatal(err)
+					}
+					if shape != "visible" && before.Stamp.Rows != 0 {
+						t.Fatal("fixture did not create an empty view", before)
+					}
+					request, err := g.PrepareProjection()
+					if err != nil {
+						t.Fatal(err)
+					}
+					oldIdentity, oldRevision, oldRows := base.identity, base.revision, base.rows
+					switch drift {
+					case "identity":
+						base.identity = "different-domain"
+					case "revision":
+						base.revision++
+					case "rows":
+						base.rows = append(append([][]string(nil), base.rows...), []string{"new", "0", "keep"})
+					}
+					if got, err := sourceStamp(view); err != nil || got != before.Stamp {
+						t.Fatal("test must retain the projection's accepted view stamp", got, before.Stamp, err)
+					}
+					operations := []struct {
+						name string
+						run  func() error
+					}{
+						{"State", func() error { _, err := g.State(); return err }},
+						{"Snapshot", func() error { _, err := g.Snapshot(); return err }},
+						{"Refresh", g.Refresh},
+						{"SetViewportNoop", func() error { return g.SetViewport(before.Viewport) }},
+						{"ScrollNoop", func() error { return g.Scroll(0, 0) }},
+						{"Move", func() error { return g.Move(1, 0) }},
+						{"Page", func() error { return g.Page(1) }},
+						{"Activate", func() error { return g.Activate(GridCellRef{}) }},
+						{"Selection", func() error { _, err := g.Selection(); return err }},
+						{"SelectAll", g.SelectAll},
+						{"BeginActiveEdit", g.BeginActiveEdit},
+						{"PrepareEdit", func() error { _, err := g.PrepareEdit(); return err }},
+						{"PrepareProjection", func() error { _, err := g.PrepareProjection(); return err }},
+					}
+					for _, op := range operations {
+						t.Run(op.name, func(t *testing.T) {
+							err := op.run()
+							if !errors.Is(err, ErrGridSource) || !errors.Is(err, ErrGridStale) {
+								t.Errorf("borrowed input drift admitted or error cause lost: %v", err)
+							}
+							if g.state() != before || g.pendingProjection != request.token {
+								t.Fatal("refusal changed accepted view or pending request")
+							}
+						})
+					}
+					base.identity, base.revision, base.rows = oldIdentity, oldRevision, oldRows
+					if got, err := g.State(); err != nil || got != before {
+						t.Fatal("refusal poisoned the original accepted source", got, err)
+					}
+					if err := g.CancelProjection(request); err != nil {
+						t.Fatal("refusal retired the original request", err)
+					}
+				})
+			}
+		}
+	}
+}

+func TestGridProjectionCurrentMetadataOnlyAndCallbackDrift(t *testing.T) {
+	base, source := projectionCurrentInput(t, 3, 0)
+	g, _ := projectionCurrentGrid(t, source, GridQuery{})
+	before, err := g.State()
+	if err != nil {
+		t.Fatal(err)
+	}
+	indexed := 0
+	base.hook = func(string, int) error {
+		indexed++
+		return errors.New("current-state inspection must not materialize indexed data")
+	}
+	for i := 0; i < 5; i++ {
+		if got, err := g.State(); err != nil || got != before {
+			t.Fatal("valid state inspection changed state or read indexed data", got, err)
+		}
+	}
+	if indexed != 0 {
+		t.Fatal("state inspection enumerated the source", indexed)
+	}
+	// Revision returns its old value, then the metadata callback changes it.
+	// A second input-stamp observation must catch that change before publication.
+	once := false
+	base.metadata = func(operation string) {
+		if operation == "revision" && !once {
+			once = true
+			base.revision++
+		}
+	}
+	if _, err := g.State(); !errors.Is(err, ErrGridSource) || !errors.Is(err, ErrGridStale) {
+		t.Fatal("metadata callback drift escaped state publication", err)
+	}
+	if !once || indexed != 0 || g.state() != before {
+		t.Fatal("metadata drift fixture did not remain bounded and atomic", once, indexed)
+	}
